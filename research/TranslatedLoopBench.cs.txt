using System.Diagnostics; using ElasticMem.Core.Access;
// Цикл приложения: сумма столбца price по SoA-буферу B. Три исполнения:
//  1) исходное: B — отдельная физическая копия (2 буфера);
//  2) без трансляции: каждое чтение B идёт через виртуальный слой GetB (аналог перехвата);
//  3) «транслированный» цикл: тот же результат, но читает поля прямо из A (AoS) с шагом — без копии B.
int n = 4 << 20, f = 4;                 // 4M записей × 4 поля = 128 МБ A
var rnd = new Random(1);
var A = new ulong[(long)n * f]; for (long i = 0; i < A.LongLength; i++) A[i] = (ulong)rnd.Next(1 << 30);
var tr = new AoSoATransform(n, f);
var Bcopy = new ulong[A.LongLength]; for (long i = 0; i < Bcopy.LongLength; i++) Bcopy[i] = A[tr.Map(i)];
var view = new LinkedView(A, tr);
int col = 1; long off = (long)col * n;
double Best(Func<(double, ulong)> fn, out ulong r) { fn(); double b = 1e18; ulong x = 0; for (int k = 0; k < 3; k++) { var (t, v) = fn(); if (t < b) b = t; x = v; } r = x; return b; }
double t1 = Best(() => { var w = Stopwatch.StartNew(); ulong s = 0; for (long i = 0; i < n; i++) s += Bcopy[off + i]; return (w.Elapsed.TotalMilliseconds, s); }, out var r1);
double t2 = Best(() => { var w = Stopwatch.StartNew(); ulong s = 0; for (long i = 0; i < n; i++) s += view.GetB(off + i); return (w.Elapsed.TotalMilliseconds, s); }, out var r2);
double t3 = Best(() => { var w = Stopwatch.StartNew(); ulong s = 0; for (long r = 0; r < n; r++) s += A[r * f + col]; return (w.Elapsed.TotalMilliseconds, s); }, out var r3);
Console.WriteLine($"Сумма столбца (4M значений): A = 128 МБ, B = 128 МБ");
Console.WriteLine($"  1) отдельная копия B (256 МБ всего):           {t1,7:F1} мс");
Console.WriteLine($"  2) виртуальный B через слой B=F(A)⊕R (128 МБ):  {t2,7:F1} мс  ({t2 / t1:F1}× медленнее)");
Console.WriteLine($"  3) транслированный цикл по A с шагом (128 МБ):  {t3,7:F1} мс  ({t3 / t1:F1}×)   совпадает: {r1 == r2 && r2 == r3}");
// случайный доступ
var idx = new long[4_000_000]; for (int i = 0; i < idx.Length; i++) idx[i] = rnd.NextInt64(A.LongLength);
double q1 = Best(() => { var w = Stopwatch.StartNew(); ulong s = 0; foreach (var i in idx) s += Bcopy[i]; return (w.Elapsed.TotalNanoseconds / idx.Length, s); }, out var a1);
double q2 = Best(() => { var w = Stopwatch.StartNew(); ulong s = 0; foreach (var i in idx) s += view.GetB(i); return (w.Elapsed.TotalNanoseconds / idx.Length, s); }, out var a2);
Console.WriteLine($"Случайное чтение B: копия {q1:F1} нс, виртуальный {q2:F1} нс; совпадает: {a1 == a2}");
